resource "aws_lambda_function" "my_lambda" {
  function_name = "my_lambda"   #

  role     = aws_iam_role.lambda_role.arn  #role
  runtime  = "python3.12"
  handler  = "lambda_function.lambda_handler"    #Code madhla entry point

  filename = "lambda.zip"      #Lambda code aslela ZIP file jo AWS var upload hoto
}

resource "aws_iam_role" "lambda_role" {
  name = "lambda-role"

  assume_role_policy = jsonencode({
    Version = "2012-10-17"
    Statement = [{
      Effect = "Allow"
      Principal = {
        Service = "lambda.amazonaws.com"
      }
      Action = "sts:AssumeRole"
    }]
  })
}
